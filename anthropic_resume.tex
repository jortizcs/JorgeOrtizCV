\section{Selected Work: Model Capabilities, Agents \& Measurement}

\project{Veritas: Epistemic Oversight for Long-Running AI Agents}{2026--Present}
{\href{https://github.com/jortizcs/veritas}{Open-source system} and
\href{https://github.com/jortizcs/veritas-experiments}{evaluation harness}}
\begin{itemize}
  \item Designed an evidence-centered framework that records agent claims, decomposes verification obligations, runs independent checks, and gates consequential actions when evidence is unresolved.
  \item Built controlled, multi-model evaluations of long-running workflows using matched conditions, clean-task do-no-harm panels, seeded faults, independent scoring, signed traces, explicit budgets and stopping rules, and reproducible evidence bundles.
  \item Treat negative results and evaluator failures as first-class findings; use fault injection and replay to distinguish genuine repair, safe blocking, unnecessary intervention, and unsupported release.
\end{itemize}

\project{TraceFix: Verified Coordination for Agentic Systems}{ACM CAIS 2026}
{Outstanding Solution Paper}
\begin{itemize}
  \item Use TLA+ model checking and counterexample-guided repair to find and fix deadlocks, races, and protocol violations in multi-agent coordination; enforce verified protocols with runtime topology monitoring.
\end{itemize}

\project{Explicit Abstention for Predictable VLM Reliability}{2026}
{\href{https://arxiv.org/abs/2601.00138}{Preprint}}
\begin{itemize}
  \item Developed explicit abstention controls for video question answering to expose capability--coverage tradeoffs and make system reliability more predictable under uncertainty.
\end{itemize}

\project{Human-Centered RLHF for Autonomous Driving}{IMWUT/UbiComp 2024}
